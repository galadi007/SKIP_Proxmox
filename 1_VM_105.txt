qm create 105 \
  --name SKIP \
  --ostype l26 \
  --sockets 1 \
  --cores 4 \
  --cpu host \
  --memory 12288 \
  --numa 0 \
  --agent 1 \
  --onboot 1 \
  --scsihw virtio-scsi-single \
  --net0 virtio=BC:24:11:29:63:8E,bridge=vmbr0,firewall=1 &&

qm set 105 \
  --scsi0 zfspool:120,iothread=1,ssd=1 &&

qm set 105 \
  --ide2 local:iso/ubuntu-24.04.2-live-server-amd64.iso,media=cdrom &&

qm set 105 \
  --boot 'order=ide2;scsi0' &&

qm config 105 &&

qm start 105 &&

qm status 105
